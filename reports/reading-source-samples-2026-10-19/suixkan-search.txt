<script>

    $(function () {
        $(".icon_back").on("click",function(){history.back()});
    });

    var tur = true;
    var range = 0,
        maxpage = parseInt( 208 ) + 1,
        page = 2,
        totalHeight = 0;

    function loadData() {
        $(window).scroll(function() {
            if(tur) {
                setTimeout(fnscroll, 100);
                tur = false;
            }
        })
    }
    loadData();

    function fnscroll() {
        if(page == -1)
            return;
        var scrollHeight = $(window).scrollTop(); 
        totalHeight = parseFloat($(window).height()) + parseFloat(scrollHeight); 
        if(page == maxpage){
            $('.v-list').append('<div class="z_loading1">已到底</div>');
            page = -1;
            return;
        }
        if(($(document).height()*2/3) <= totalHeight && page != maxpage) {
            if($(document).height() <= totalHeight)
                $('.v-list').append('<div class="z_loading"><em/>正在加载更多数据</div>');
            $.ajax({
                async : true,
                cache : false,
                type : 'get',
                url : '/api/book/search?keyword=三国' + '&page=' + page,
                dataType : "json",
                error : function() {},
                success : function(result) {
                    $(".z_loading").remove();
                    $(".z_loading1").remove();
                    if(result.code == 0){
                        if(result.data.length == 0){
                            $('.v-list').append('<div class="z_loading1">已到底</div>');
                            page = -1;
                        }else{
                            addData(result);
                            page++;
                        }
                    }
                }
            });

        }
        tur = true;
    }

    function addData(result) {
        for(var i = 0; i < result.data.list.length; i++) {
            var book = result.data.list[i];
            var str = '/b/' + book.wapBookId+'.html';
            var s='<div class="v-list-item flex" onclick=\"gotoPage(\''+str+'\', \'\',\'\')\">'+
                '<div class="v-cover"> <img class="v-cover-img" src='+book.bookPic+'> </div>' +
                '<div class="v-info flex-item flexcol-justify">' +
                '<div class="v-content">' +
                '<div class="v-title">'+book.bookName+'</div>' +
                '<div class="v-intro">'+book.intro+'</div></div>' +
                '<div class="v-other flex-v-end">' +
                '<div class="v-author">'+book.authorName+'&nbsp; &nbsp;</div>' +
                '<div class="v-words flex-item">'+(book.words/10000).toFixed(1)+'万字</div></div></div>';
            $('.v-list').append(s);
        }
    }
</script>
