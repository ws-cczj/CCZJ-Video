    var maxPage = "16";

    function selectChange() {
        var objS = document.getElementById("chapterFilter");

        document.getElementById(value1).style.display = "";

        ctrl_button(value);

    function pre_page() {
        var cur = parseInt(value);
        var pre_cur = cur - 1;
        if (pre_cur >= 1) {
            $("#next_button_id_box").show();
            $("#chapterFilter").val(pre_cur);
            ctr_page(pre_cur, cur);
            value = pre_cur;
            if (pre_cur == 1) {
                $("#prev_button_id_box").hide();
                $("#next_button_id_box").show();
            }
        }
    }

    function next_page() {
        var cur = parseInt(value);
        var next_cur = cur + 1;
        if (next_cur <= parseInt(maxPage)) {
            $("#prev_button_id_box").show();
            $("#chapterFilter").val(cur);
            ctr_page(next_cur, cur);
            value = next_cur;
            if (cur == parseInt(maxPage)) {
                $("#next_button_id_box").hide();
            }
        }
    }

    function ctrl_button(curr_page) {

            $("#next_button_id_box").show();
