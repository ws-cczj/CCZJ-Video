  function scheduleAdAlignment() {
    if (frame) return;
    frame = requestAnimationFrame(alignAdToVisualBottom);
  }

  new MutationObserver(scheduleAdAlignment).observe(document.body, {
    childList: true,
    subtree: true
  });
  if (visualViewport) {
    visualViewport.addEventListener('resize', scheduleAdAlignment, { passive: true });
    visualViewport.addEventListener('scroll', scheduleAdAlignment, { passive: true });
  }
  window.addEventListener('resize', scheduleAdAlignment, { passive: true });
  window.addEventListener('orientationchange', scheduleAdAlignment, { passive: true });
  scheduleAdAlignment();
  [250, 1000, 3000, 6000].forEach(function (delay) {
    window.setTimeout(scheduleAdAlignment, delay);
  });
})();
