func pixel(c color.RGBA) []byte {
 var output bytes.Buffer
 img := image.NewRGBA(image.Rect(0,0,1,1)); img.SetRGBA(0,0,c)
 if err := png.Encode(&output,img); err != nil { panic(err) }
 return output.Bytes()
}

func main() {
 dir,err := os.MkdirTemp("", "cczj-extension-audit-"); if err != nil { panic(err) }; defer os.RemoveAll(dir)
 if err := applog.Init(dir); err != nil { panic(err) }; defer applog.Default().Close()
 proxy.SetAllowPrivateTargets(true); defer proxy.SetAllowPrivateTargets(false)
 a,b := pixel(color.RGBA{255,0,0,255}),pixel(color.RGBA{0,0,255,255})
 var requests atomic.Int32
 server := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter,r *http.Request) {
  requests.Add(1); w.Header().Set("Content-Type","image/png")
  if r.Header.Get("X-Variant")=="source_b" { _,_=w.Write(b) } else { _,_=w.Write(a) }
 })); defer server.Close()
 images := proxy.NewImageProxy(func()string{return dir})
 images.SetHeaderResolver(func(key string)map[string]string{return map[string]string{"X-Variant":key}})
 fetch := func(p *proxy.ImageProxy,upstream,key string) *httptest.ResponseRecorder {
  response:=httptest.NewRecorder(); p.ServeHTTP(response,httptest.NewRequest(http.MethodGet,p.URL(upstream,"",key),nil)); return response
 }
 first:=fetch(images,server.URL,"source_a"); second:=fetch(images,server.URL,"source_b")
 result:=map[string]any{
  "source_keys":[]string{model.DeriveSourceKey("https://api.example.com/novel"),model.DeriveSourceKey("https://api.example.com/comic"),model.DeriveSourceKey("https://example.cn/api")},
  "image_context":map[string]any{"first_status":first.Code,"second_status":second.Code,"upstream_requests":requests.Load(),"second_received_source_b":bytes.Equal(second.Body.Bytes(),b),"second_reused_source_a":bytes.Equal(second.Body.Bytes(),a)},
 }
 var coldRequests atomic.Int32
 gate:=make(chan struct{})
 coldServer:=httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter,r *http.Request){
  if coldRequests.Add(1)==8 {close(gate)}
  select {case <-gate: case <-time.After(time.Second):}
  w.Header().Set("Content-Type","image/png"); _,_=w.Write(a)
 })); defer coldServer.Close()
 cold:=proxy.NewImageProxy(func()string{return dir+"/cold"})
 var group sync.WaitGroup
 var failures atomic.Int32
 for i:=0;i<8;i++ {group.Add(1); go func(){defer group.Done(); if fetch(cold,coldServer.URL,"same_source").Code!=200 {failures.Add(1)}}()}
 group.Wait()
 result["duplicate_cold_reads"]=map[string]any{"callers":8,"upstream_requests":coldRequests.Load(),"failures":failures.Load()}
 encoder:=json.NewEncoder(os.Stdout); encoder.SetIndent("","  "); if err:=encoder.Encode(result);err!=nil{panic(err)}
}
