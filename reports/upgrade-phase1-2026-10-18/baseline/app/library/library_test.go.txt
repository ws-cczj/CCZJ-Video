// 这一组是端到端测试：真的起一个本地源站、真的走 provider 的提取引擎、
// 真的读写 SQLite。分层单测证明不了"整条链路接通"，而这条链路上每一环
// 单独看都是对的、合起来却可能断在字段名对不上——那正是最难查的一类故障。

// novelDocument 是一份能跑通全链路的小说源文档。
// 它对着下面 fakeSource 返回的 HTML 写，两者必须一起改。
func novelDocument() string {
	return `{
		"version": 3,
		"kind": "novel",
		"check_keyword": "示例",
		"operations": {
			"search": {
				"format": "html",
				"list": {"query": "ul.results > li"},
				"fields": {
					"title": {"query": "a", "attr": "text"},
					"url": {"query": "a", "attr": "href"},
					"author": {"query": "span.author"}
				}
			},
			"detail": {
				"format": "html",
				"root": {"query": "div.detail"},
				"fields": {
					"title": {"query": "h1"},
					"author": {"query": "p.author"},
					"summary": {"query": "p.intro"},
					"toc_url": {"query": "a.toc", "attr": "href"}
				}
			},
			"toc": {
				"format": "html",
				"list": {"query": "ul.chapters > li"},
				"fields": {
					"title": {"query": "a", "attr": "ownText"},
					"chapter_url": {"query": "a", "attr": "href"}
				}
			},
			"content": {
				"format": "html",
				"text": {"query": "div#content", "attr": "text"}
			}
		}
	}`
}

// fakeSource 是一个形状真实的小说站：搜索页、详情页、目录页、正文页。
// 目录里故意夹一个 VIP 标记的 <span>，用来验证 ownText 不会把它吞进标题。
func fakeSource() *httptest.Server {
	return httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		w.Header().Set("Content-Type", "text/html; charset=utf-8")
		switch {
		// 搜索是根路径 + query，与 MAC-CMS 形状一致：源站不会为搜索单开一条路径。
		case r.URL.Query().Get("wd") != "":
			fmt.Fprint(w, `<html><body><ul class="results">
				<li><a href="/book/1">示例书</a><span class="author">作者甲</span></li>
				<li><a href="/book/2">另一本</a><span class="author">作者乙</span></li>
			</ul></body></html>`)
		case r.URL.Path == "/book/1" || r.URL.Path == "/book/2":
			fmt.Fprintf(w, `<html><body><div class="detail">
				<h1>书%s</h1><p class="author">作者</p><p class="intro">简介</p>
				<a class="toc" href="%s/toc">目录</a></div></body></html>`,
				strings.TrimPrefix(r.URL.Path, "/book/"), r.URL.Path)
		case strings.HasSuffix(r.URL.Path, "/toc"):
			fmt.Fprint(w, `<html><body><ul class="chapters">
				<li class="vol">第一卷</li>
				<li><a href="/c/1.html">第一章 起风<span class="vip">VIP</span></a></li>
				<li><a href="/c/2.html">第二章 落雨</a></li>
			</ul></body></html>`)
		case strings.HasPrefix(r.URL.Path, "/c/"):
			fmt.Fprint(w, `<html><body><div id="content"><p>第一段正文。</p><p>第二段正文。</p></div></body></html>`)
		default:
			http.NotFound(w, r)
		}
	}))
}

// testDataDir 与 testCacheDir 在整个测试进程里是固定的：db.InitDB 受 sync.Once
// 保护，一个进程只能初始化一次，所以各测试共享一个库、各自用不同的 source_key
// 并在开始前清掉自己那一份。缓存目录同样必须稳定——正文缓存命中与否是靠磁盘上
// 那一份文件判定的，每个测试换一个目录会让"第二次取正文应当命中缓存"永远失败，
// 而且失败原因看不出来。
var (
	testDataDir  string
	testCacheDir string
)

func TestMain(m *testing.M) {
	// httptest 绑 127.0.0.1，而 SSRF 闸门默认拒绝私网地址。这不是绕过校验，
	// 是用它自带的用户开关（自建源与 NAS 靠它才连得上）。
	proxy.SetAllowPrivateTargets(true)
	root, err := os.MkdirTemp("", "cczj-library-test-")
	if err != nil {
		fmt.Fprintln(os.Stderr, err)
		os.Exit(1)
	}
	testDataDir = filepath.Join(root, "data")
	testCacheDir = filepath.Join(root, "cache")
	if err := db.InitDB(testDataDir); err != nil {
		fmt.Fprintln(os.Stderr, err)
		os.Exit(1)
	}
	code := m.Run()
	// 先关库再删目录：Windows 上开着句柄删不掉文件，
	// 而删不掉会让 t.TempDir 那类清理报一个和被测逻辑无关的错。
	db.Close()
	proxy.SetAllowPrivateTargets(false)
	_ = os.RemoveAll(root)
	os.Exit(code)
}

func setupLibraryTest(t *testing.T) *Service {
	t.Helper()
	// 库是整个测试进程共享的，而 Search 会遍历**所有**启用的同类源：
	// 不清干净的话上一个测试留下的源（指向一个已经 Close 的 httptest）
	// 会让这一个测试的搜索结果里多出一条"连接被拒"，
	// 表现出来是"got 3 results, want 2"这种与被测逻辑无关的失败。
	for _, kind := range []model.SourceKind{model.SourceKindNovel, model.SourceKindComic} {
		sources, err := db.GetSourcesByKind(kind)
		if err != nil {
			t.Fatal(err)
		}
		for _, source := range sources {
			cleanupTestSource(t, source.SourceKey)
		}
	}
	return NewService(func() string { return testCacheDir })
}

func addTestSource(t *testing.T, server *httptest.Server, key, document string) {
	t.Helper()
	cleanupTestSource(t, key)
	row := &model.Source{
		SourceKey: key, Name: key, ApiUrl: server.URL, Enabled: 1,
		Kind: string(model.SourceKindNovel), StrategyConfig: document,
	}
	if err := db.AddSource(row); err != nil {
		t.Fatal(err)
	}
}

// cleanupTestSource 清掉一个 source_key 的全部痕迹。
//
// 库是整个测试进程共享的（db.InitDB 受 sync.Once 保护，没法每个测试重开），
// 所以每个测试开头都要先清一遍自己那份，否则会读到上一个测试留下的行。
func cleanupTestSource(t *testing.T, key string) {
	t.Helper()
	_, _ = db.DB().Exec(`DELETE FROM library_chapters WHERE item_id IN (SELECT id FROM library_items WHERE source_key=?)`, key)
	_, _ = db.DB().Exec(`DELETE FROM library_items WHERE source_key=?`, key)
	_, _ = db.DB().Exec(`DELETE FROM sources WHERE source_key=?`, key)
}

func TestSearchReturnsItemsAndPersistsThem(t *testing.T) {
	service := setupLibraryTest(t)
	server := fakeSource()
	defer server.Close()
	addTestSource(t, server, "fake_novel", novelDocument())

	outcome, err := service.Search(context.Background(), model.SourceKindNovel, "示例", 1, nil)
	if err != nil {
		t.Fatal(err)
	}
	if outcome.FailedCount != 0 {
		t.Fatalf("failed count = %d, want 0; first result: %+v", outcome.FailedCount, func() any {
			if len(outcome.Results) > 0 {
				r := *outcome.Results[0]
				return &r
			}
			return nil
		}())
	}
	if len(outcome.Results) != 1 {
		t.Fatalf("got %d source results, want 1", len(outcome.Results))
	}
	result := outcome.Results[0]
	if result.ErrorCode != "" {
		t.Fatalf("source failed: %s: %s", result.ErrorCode, result.ErrorMessage)
	}
	if len(result.Items) != 2 {
		t.Fatalf("got %d items, want 2", len(result.Items))
	}
	first := result.Items[0]
	if first.Title != "示例书" || first.Author != "作者甲" {
		t.Fatalf("item = %+v", first)
	}
	if first.ID == 0 {
		t.Fatal("搜索结果没有落库：拿不到本地 id，详情页就没法关联")
	}
	if first.SourceKey != "fake_novel" || first.RemoteID == "" {
		t.Fatalf("item identity is incomplete: %+v", first)
	}

	// 落库之后要能按 id 取回来，且目录同步能跑通。
	detail, err := service.Detail(context.Background(), first.ID, true)
	if err != nil {
		t.Fatal(err)
	}
	if detail.Item == nil || detail.Item.Title == "" {
		t.Fatalf("detail = %+v", detail.Item)
	}
	// 目录里那条"第一卷"没有 <a>，标题取不到，必须被筛掉；
	// 而"第一章 起风"后面的 <span>VIP</span> 必须被 ownText 排除。
	active := 0
	for _, chapter := range detail.Chapters {
		if chapter.Available {
			active++
		}
		if strings.Contains(chapter.Title, "VIP") {
			t.Fatalf("ownText 泄漏了嵌套的 span：%q", chapter.Title)
		}
	}
	if active != 2 {
		t.Fatalf("active chapters = %d, want 2 (卷标题行应当被筛掉)", active)
	}

	if len(detail.Chapters) == 0 {
		t.Fatalf("目录是空的；详情=%+v", detail.Item)
	}
	// 正文：第一次抓，第二次命中缓存。
	var chapterID int
	for _, chapter := range detail.Chapters {
		if chapter.Available && !chapter.IsVolume {
			chapterID = chapter.ID
			break
		}
	}
	if chapterID == 0 {
		t.Fatalf("没有可用的章节；chapters=%+v", detail.Chapters)
	}
	text, err := service.ChapterText(context.Background(), chapterID)
	if err != nil {
		t.Fatal(err)
	}
	if text.FromCache {
		t.Fatal("第一次取正文不该命中缓存")
	}
	if strings.Join(text.Paragraphs, "|") != "第一段正文。|第二段正文。" {
		t.Fatalf("paragraphs = %q", text.Paragraphs)
	}
	again, err := service.ChapterText(context.Background(), chapterID)
	if err != nil {
		t.Fatal(err)
	}
	if !again.FromCache {
		t.Fatal("第二次取正文应当命中磁盘缓存，否则每次翻页都重新出网")
	}
}

func TestSearchReportsPerSourceFailureWithoutEmptyingTheRest(t *testing.T) {
	service := setupLibraryTest(t)
	good := fakeSource()
	defer good.Close()
	// 坏源：一切请求都返回 500。
	bad := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		http.Error(w, "boom", http.StatusInternalServerError)
	}))
	defer bad.Close()

	addTestSource(t, good, "good_novel", novelDocument())
	addTestSource(t, bad, "bad_novel", novelDocument())

	outcome, err := service.Search(context.Background(), model.SourceKindNovel, "示例", 1, nil)
	if err != nil {
		t.Fatal(err)
	}
	// 一个源挂了不该让整次搜索失败，也不该让另一个源的结果消失。
	if len(outcome.Results) != 2 {
		t.Fatalf("got %d results, want 2 (每个源各自成功或失败)", len(outcome.Results))
	}
	if outcome.FailedCount != 1 {
		t.Fatalf("failed count = %d, want 1", outcome.FailedCount)
	}
	var goodResult, badResult *SourceResult
	for _, result := range outcome.Results {
		if result.SourceKey == "good_novel" {
			goodResult = result
		}
		if result.SourceKey == "bad_novel" {
			badResult = result
		}
	}
	if goodResult == nil || len(goodResult.Items) != 2 {
		t.Fatalf("好源的结果被坏源带崩了：%+v", goodResult)
	}
	if goodResult.ErrorCode != "" {
		t.Fatalf("好源不该带错误码：%s", goodResult.ErrorCode)
	}
	if badResult == nil || badResult.ErrorCode == "" {
		t.Fatalf("坏源必须带错误码，否则界面上它与'确实没有这本书'分不开：%+v", badResult)
	}
	if len(badResult.Items) != 0 {
		t.Fatalf("坏源不该有条目：%+v", badResult.Items)
	}
}

func TestEmptySearchIsASuccessNotAnError(t *testing.T) {
	service := setupLibraryTest(t)
	// 返回一个结构正确但一条结果都没有的搜索页。
	server := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		w.Header().Set("Content-Type", "text/html; charset=utf-8")
		fmt.Fprint(w, `<html><body><ul class="results"></ul></body></html>`)
	}))
	defer server.Close()
	addTestSource(t, server, "empty_novel", novelDocument())

	outcome, err := service.Search(context.Background(), model.SourceKindNovel, "不存在的书", 1, nil)
	if err != nil {
		t.Fatal(err)
	}
	result := outcome.Results[0]
	// 空列表只能代表一次成功的"没有结果"（reports/00:135）。
	// 如果这里带上了错误码，界面上就会把"这个源确实没有这本书"显示成"搜索失败"，
	// 而两者的处置完全相反：前者该换个词或换个源，后者该重试。
	if result.ErrorCode != "" {
		t.Fatalf("一次成功的空结果不该带错误码：%s: %s", result.ErrorCode, result.ErrorMessage)
	}
	if len(result.Items) != 0 {
		t.Fatalf("items = %+v, want empty", result.Items)
	}
	if outcome.FailedCount != 0 {
		t.Fatalf("failed count = %d, want 0", outcome.FailedCount)
	}
}

func TestAddSourceRejectsUnsupportedFields(t *testing.T) {
	service := setupLibraryTest(t)
	server := fakeSource()
	defer server.Close()
	// httptest 的地址是 127.0.0.1:随机端口，DeriveSourceKey 会从主机名推出
	// 一个固定的 key（"127"），所以每个用 AddSource 的测试都得先清掉它，
	// 否则上一个测试留下的行会让这一个报"来源已存在"。
	derived := model.DeriveSourceKey(server.URL)
	cleanupTestSource(t, derived)

	// 一份为别的引擎写的文档，带着 js 字段。必须在**写入之前**报出具名错误：
	// 静默忽略的话用户看到的就是"源加上了但搜不出东西"，
	// 与"源真的没有这本书"无法区分。
	withJS := strings.Replace(novelDocument(), `"check_keyword": "示例",`, `"check_keyword": "示例", "js": "alert(1)",`, 1)
	if _, err := service.AddSource(model.SourceKindNovel, "坏源", server.URL, withJS); err == nil {
		t.Fatal("带 js 字段的文档被接受了")
	} else if !strings.Contains(err.Error(), "js") {
		t.Fatalf("报错没有指名是哪个字段：%v", err)
	}
	// 被拒的源不该落库：否则源列表里会多一个永远不能用的幽灵源。
	if existing, _ := db.GetSourceByKey(derived); existing != nil {
		t.Fatal("被拒的源落库了")
	}

	if _, err := service.AddSource(model.SourceKindNovel, "好源", server.URL, novelDocument()); err != nil {
		t.Fatalf("合法文档被拒了：%v", err)
	}
	// 视频源不走这个入口：它有自己的一套采集配置与调度。
	if _, err := service.AddSource(model.SourceKindVideo, "视频", server.URL, "{}"); err == nil {
		t.Fatal("视频源不该能从阅读模式的入口加进来")
	}
}

func TestDetailSurvivesAMissingSource(t *testing.T) {
	service := setupLibraryTest(t)
	server := fakeSource()
	defer server.Close()
	addTestSource(t, server, "doomed_novel", novelDocument())

	outcome, err := service.Search(context.Background(), model.SourceKindNovel, "示例", 1, nil)
	if err != nil {
		t.Fatal(err)
	}
	itemID := outcome.Results[0].Items[0].ID

	// 源被删掉，但书还在书架上。
	if err := db.DeleteSource("doomed_novel"); err != nil {
		t.Fatal(err)
	}
	detail, err := service.Detail(context.Background(), itemID, false)
	if err != nil {
		t.Fatalf("源没了就该整页报错：%v", err)
	}
	// 必须仍然返回条目本身并标记不可用：书架指着它，
	// 让它凭空消失会让用户以为数据丢了。
	if detail.Item == nil {
		t.Fatal("detail.Item 是 nil，书架上这一行会消失")
	}
	if detail.Item.Available {
		t.Fatal("源已删除，条目的 available 必须是 false，界面才能显示成「来源已下架」")
	}
	if detail.Item.Title == "" {
		t.Fatalf("条目丢了标题：%+v", detail.Item)
	}
}

func TestTextCacheIsInvalidatedByDocumentVersion(t *testing.T) {
	service := setupLibraryTest(t)
	hits := 0
	server := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		w.Header().Set("Content-Type", "text/html; charset=utf-8")
		switch {
		// 搜索是根路径 + query，与 MAC-CMS 形状一致：源站不会为搜索单开一条路径。
		case r.URL.Query().Get("wd") != "":
			fmt.Fprint(w, `<html><body><ul class="results"><li><a href="/book/1">示例书</a></li></ul></body></html>`)
		case strings.HasSuffix(r.URL.Path, "/toc"):
			fmt.Fprint(w, `<html><body><ul class="chapters"><li><a href="/c/1.html">第一章</a></li></ul></body></html>`)
		case strings.HasPrefix(r.URL.Path, "/book/"):
			fmt.Fprint(w, `<html><body><div class="detail"><h1>书</h1><a class="toc" href="/book/1/toc">目录</a></div></body></html>`)
		case strings.HasPrefix(r.URL.Path, "/c/"):
			hits++
			// 每次抓都返回不同的正文，这样"有没有重新出网"是可观测的。
			fmt.Fprintf(w, `<html><body><div id="content"><p>第 %d 次抓取</p></div></body></html>`, hits)
		}
	}))
	defer server.Close()
	addTestSource(t, server, "cached_novel", novelDocument())

	outcome, err := service.Search(context.Background(), model.SourceKindNovel, "示例", 1, nil)
	if err != nil {
		t.Fatal(err)
	}
	detail, err := service.Detail(context.Background(), outcome.Results[0].Items[0].ID, true)
	if err != nil {
		t.Fatal(err)
	}
	chapterID := detail.Chapters[0].ID

	first, err := service.ChapterText(context.Background(), chapterID)
	if err != nil {
		t.Fatal(err)
	}
	if first.FromCache || hits != 1 {
		t.Fatalf("first read: fromCache=%v hits=%d, want false/1", first.FromCache, hits)
	}
	second, err := service.ChapterText(context.Background(), chapterID)
	if err != nil {
		t.Fatal(err)
	}
	if !second.FromCache || hits != 1 {
		t.Fatalf("second read: fromCache=%v hits=%d, want true/1（缓存没生效，每次翻页都会重新出网）", second.FromCache, hits)
	}

	// 改了源文档就必须作废旧缓存：否则用户读到的是旧规则解析出来的正文，
	// 而界面上看不出来（ADR 0016 把源文档指纹放进了缓存键，就是为了这件事）。
	changed := strings.Replace(novelDocument(), `"attr": "text"`, `"attr": "ownText"`, 1)
	if err := service.UpdateSource("cached_novel", "", "", changed); err != nil {
		t.Fatal(err)
	}
	third, err := service.ChapterText(context.Background(), chapterID)
	if err != nil {
		t.Fatal(err)
	}
	if third.FromCache {
		t.Fatal("改了源文档之后仍然命中缓存：缓存键里没有源文档指纹")
	}
	if hits != 2 {
		t.Fatalf("hits = %d, want 2（改配置后应当重新抓一次）", hits)
	}
}

func TestTestSourceReportsEveryBrokenStage(t *testing.T) {
	service := setupLibraryTest(t)
	// 目录规则指向一个不存在的选择器：搜索与详情能通，目录会空。
	brokenToc := strings.Replace(novelDocument(), `"query": "ul.chapters > li"`, `"query": "ul.nonexistent > li"`, 1)
	server := fakeSource()
	defer server.Close()
	cleanupTestSource(t, model.DeriveSourceKey(server.URL))
	if _, err := service.AddSource(model.SourceKindNovel, "分级测试", server.URL, brokenToc); err != nil {
		t.Fatal(err)
	}

	outcome, err := service.TestSource(context.Background(), model.DeriveSourceKey(server.URL))
	if err != nil {
		t.Fatal(err)
	}
	byName := map[string]*TestStage{}
	for _, stage := range outcome.Stages {
		byName[stage.Name] = stage
	}
	// 逐级测试的意义在于：搜索通了不代表能读。只报一个总的成功/失败，
	// 用户就得自己猜是哪一步坏了。
	for _, name := range []string{StageConfig, StageSearch, StageDetail, StageToc} {
		if byName[name] == nil {
			t.Fatalf("缺少 %q 这一步：%+v", name, outcome.Stages)
		}
	}
	if byName[StageSearch].Status != "ok" || byName[StageSearch].Count != 2 {
		t.Fatalf("搜索这一步应当成功并取到 2 条：%+v", byName[StageSearch])
	}
	if byName[StageToc].Status != "failed" {
		t.Fatalf("目录规则写坏了，这一步必须报失败，而不是让整体显示成「源可用」：%+v", byName[StageToc])
	}
	if byName[StageToc].Millis < 0 {
		t.Fatalf("耗时不该是负数：%+v", byName[StageToc])
	}
}

func TestDetailUsesTheStoredURLNotARebuiltOne(t *testing.T) {
	service := setupLibraryTest(t)
	var detailPaths []string
	server := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		w.Header().Set("Content-Type", "text/html; charset=utf-8")
		switch {
		case r.URL.Query().Get("wd") != "":
			// 这个源声明了显式的 id 字段，所以 remote_id 会是 "b-1" 这种
			// **不是路径**的值。详情页地址带一段 /reader/ 前缀，
			// 与 api_url + remote_id 重建出来的 /b-1 可以区分开。
			fmt.Fprint(w, `<html><body><ul class="results">
				<li><a href="/reader/b-1">示例书</a><i>b-1</i></li>
			</ul></body></html>`)
		default:
			detailPaths = append(detailPaths, r.URL.Path)
			fmt.Fprint(w, `<html><body><div class="detail"><h1>书</h1>
				<a class="toc" href="/reader/b-1/toc">目录</a></div></body></html>`)
		}
	}))
	defer server.Close()

	document := strings.Replace(novelDocument(),
		`"title": {"query": "a", "attr": "text"},
					"url": {"query": "a", "attr": "href"},
					"author": {"query": "span.author"}`,
		`"id": {"query": "i"},
					"title": {"query": "a", "attr": "text"},
					"url": {"query": "a", "attr": "href"}`, 1)
	addTestSource(t, server, "explicit_id", document)

	outcome, err := service.Search(context.Background(), model.SourceKindNovel, "示例", 1, nil)
	if err != nil {
		t.Fatal(err)
	}
	if len(outcome.Results) != 1 || len(outcome.Results[0].Items) != 1 {
		t.Fatalf("搜索结果不对：%+v", outcome.Results)
	}
	item := outcome.Results[0].Items[0]
	// remote_id 用的是源声明的显式 id，不是 URL 推出来的路径。
	if item.RemoteID != "b-1" {
		t.Fatalf("remote_id = %q, want b-1（源声明了显式 id 字段）", item.RemoteID)
	}

	if _, err := service.Detail(context.Background(), item.ID, true); err != nil {
		t.Fatal(err)
	}
	// 这一条是这个测试存在的理由：详情请求必须打到存下来的 remote_url，
	// 而不是用 api_url + remote_id 重建出来的地址。
	// 重建的话会得到 /b-1；正确的是 /reader/b-1。
	if len(detailPaths) == 0 {
		t.Fatal("详情请求没有发出去")
	}
	if detailPaths[0] != "/reader/b-1" {
		t.Fatalf("详情请求打到了 %q, want /reader/b-1（打到 /b-1 说明用了重建地址而不是存下来的 remote_url）", detailPaths[0])
	}
}
