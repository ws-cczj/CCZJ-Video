// allowPrivateForTest 打开 SSRF 闸门里的私网开关。
//
// 这不是绕过校验：ValidateTarget 默认拒绝回环与私网地址，而测试里的上游
// 一律是打桩的 127.0.0.1（transport 被换成 roundTripperFunc，不会真的连出去）。
// 用的是它自带的那个用户开关——自建源与 NAS 靠它才连得上。
func allowPrivateForTest(t *testing.T) {
	t.Helper()
	previous := AllowPrivateTargets()
	SetAllowPrivateTargets(true)
	t.Cleanup(func() { SetAllowPrivateTargets(previous) })
}

const htmlErrorPage = `<html><head><title>403 Forbidden</title></head><body>防盗链检查未通过</body></html>`

func TestVerifyImageBytesRejectsNonImages(t *testing.T) {
	cases := []struct {
		name     string
		body     []byte
		declared string
		wantOK   bool
	}{
		// 这一条是整个测试文件存在的理由：Mihon 的下载/缓存层**不做任何内容类型
		// 校验**，HTML 错误页原样写盘然后标 Page.State.Ready，mime 不认识且魔数
		// 嗅探失败时还默认当 "jpg"（ChapterCache.kt:157-191、ImageUtil.kt:68-71），
		// 于是错误被推迟到解码期才在阅读器里暴露，而那份坏字节已经进了缓存、
		// 下次还是它。reports/00:146 明确要求"不缓存错误页为图片"。
		{"HTML 错误页声明成 html", []byte(htmlErrorPage), "text/html; charset=utf-8", false},
		{"HTML 错误页谎报成图片", []byte(htmlErrorPage), "image/jpeg", false},
		{"HTML 错误页没有类型", []byte(htmlErrorPage), "", false},
		{"JSON 错误体", []byte(`{"code":403,"msg":"forbidden"}`), "application/json", false},
		{"空响应", nil, "image/png", false},
		{"真 PNG 声明成 png", tinyPNG, "image/png", true},
		{"真 PNG 声明成 octet-stream", tinyPNG, "application/octet-stream", true},
		{"真 PNG 没有声明", tinyPNG, "", true},
	}
	for _, tc := range cases {
		contentType, err := verifyImageBytes(tc.body, tc.declared)
		if tc.wantOK && err != nil {
			t.Errorf("%s: 应当通过，却报错 %v", tc.name, err)
			continue
		}
		if !tc.wantOK {
			if err == nil {
				t.Errorf("%s: 应当被拒，却通过了（contentType=%q）", tc.name, contentType)
			}
			continue
		}
		if !strings.HasPrefix(contentType, "image/") {
			t.Errorf("%s: contentType = %q, want an image type", tc.name, contentType)
		}
	}
}

// stubImageUpstream 让 ImageProxy 的出网路径返回固定内容，并记下收到的请求头。
func stubImageUpstream(t *testing.T, proxyService *ImageProxy, status int, contentType string, payload []byte) *[]*http.Request {
	t.Helper()
	var seen []*http.Request
	proxyService.client.Transport = roundTripperFunc(func(request *http.Request) (*http.Response, error) {
		seen = append(seen, request)
		header := http.Header{}
		if contentType != "" {
			header.Set("Content-Type", contentType)
		}
		return &http.Response{
			StatusCode: status,
			Header:     header,
			Body:       io.NopCloser(bytes.NewReader(payload)),
		}, nil
	})
	return &seen
}

func imageRequest(upstream, referer, sourceKey string) *http.Request {
	target := "/__cczj/img?u=" + base64.RawURLEncoding.EncodeToString([]byte(upstream))
	if referer != "" {
		target += "&ref=" + base64.RawURLEncoding.EncodeToString([]byte(referer))
	}
	if sourceKey != "" {
		target += "&k=" + sourceKey
	}
	return httptest.NewRequest(http.MethodGet, target, nil)
}

func TestImageProxyKeepsTheFullReferer(t *testing.T) {
	allowPrivateForTest(t)
	dir := t.TempDir()
	service := NewImageProxy(func() string { return dir })
	seen := stubImageUpstream(t, service, http.StatusOK, "image/png", tinyPNG)

	// 完整页面地址（含路径与 query），不是 origin。
	// 截成 origin 正是 HLS 那条路由的做法（RefererOrigin，hls.go:110-116），
	// 而真实漫画 CDN 校验的往往就是含路径的那个地址——这也是新开这条路由的理由。
	referer := "http://127.0.0.1:9/book/1/2.html?page=3"
	recorder := httptest.NewRecorder()
	service.ServeHTTP(recorder, imageRequest("http://127.0.0.1:9/img/2.png", referer, ""))

	if recorder.Code != http.StatusOK {
		t.Fatalf("status = %d, want 200; body=%s", recorder.Code, recorder.Body.String())
	}
	if len(*seen) != 1 {
		t.Fatalf("upstream got %d requests, want 1", len(*seen))
	}
	if got := (*seen)[0].Header.Get("Referer"); got != referer {
		t.Fatalf("Referer = %q, want the full page address %q", got, referer)
	}
	if got := recorder.Header().Get("Content-Type"); got != "image/png" {
		t.Fatalf("Content-Type = %q, want image/png", got)
	}
}

func TestImageProxyAppliesSourceHeadersButNotRefererOrEncoding(t *testing.T) {
	allowPrivateForTest(t)
	dir := t.TempDir()
	service := NewImageProxy(func() string { return dir })
	service.SetHeaderResolver(func(sourceKey string) map[string]string {
		if sourceKey != "comic_src" {
			return nil
		}
		return map[string]string{
			"X-Site-Token":    "abc",
			"User-Agent":      "CustomAgent/1.0",
			"Referer":         "http://127.0.0.1:8/always",
			"Accept-Encoding": "br",
		}
	})
	seen := stubImageUpstream(t, service, http.StatusOK, "image/png", tinyPNG)

	recorder := httptest.NewRecorder()
	service.ServeHTTP(recorder, imageRequest("http://127.0.0.1:9/a.png", "http://127.0.0.1:9/book/9/", "comic_src"))
	if recorder.Code != http.StatusOK {
		t.Fatalf("status = %d; body=%s", recorder.Code, recorder.Body.String())
	}
	header := (*seen)[0].Header
	if got := header.Get("X-Site-Token"); got != "abc" {
		t.Fatalf("源声明的自定义头没有带上：X-Site-Token = %q", got)
	}
	if got := header.Get("User-Agent"); got != "CustomAgent/1.0" {
		t.Fatalf("User-Agent = %q, want the source's own value", got)
	}
	// Referer 必须由当前这一页决定，不能被源配置钉死：
	// 钉死之后整本漫画的每一页都会用同一个引用页，防盗链反而全挂。
	if got := header.Get("Referer"); got != "http://127.0.0.1:9/book/9/" {
		t.Fatalf("Referer = %q, want the per-page address; 源配置不该能钉死它", got)
	}
	// Accept-Encoding 一旦被声明，Go 就不再透明解压，而手动解压要求尺寸上限
	// 作用在解压后的流上。所以这个头不接受覆盖。
	if got := header.Get("Accept-Encoding"); got == "br" {
		t.Fatal("Accept-Encoding 不该被源配置覆盖")
	}
}

func TestImageProxyRefusesToCacheAnErrorPage(t *testing.T) {
	allowPrivateForTest(t)
	dir := t.TempDir()
	service := NewImageProxy(func() string { return dir })

	// 第一次：上游回一个 HTML 错误页。
	requests := stubImageUpstream(t, service, http.StatusOK, "text/html; charset=utf-8", []byte(htmlErrorPage))
	recorder := httptest.NewRecorder()
	service.ServeHTTP(recorder, imageRequest("http://127.0.0.1:9/bad.png", "", ""))
	if recorder.Code == http.StatusOK {
		t.Fatal("HTML 错误页被当成图片端出去了")
	}
	if got := requests; len(*got) != 1 {
		t.Fatalf("upstream got %d requests, want 1", len(*got))
	}

	// 关键断言：那份 HTML **没有进磁盘缓存**。
	// 进了的话，之后即使上游修好了，用户也还是会拿到那份坏字节，
	// 而界面上看不出任何区别——这正是 Mihon 那条路径的问题。
	store := newBlobStore(func() string { return dir }, "readingcache", readingCacheTTL, readingCacheMaxBytes)
	if data, contentType, ok := store.read("http://127.0.0.1:9/bad.png"); ok {
		t.Fatalf("错误页被缓存了：contentType=%q bytes=%d", contentType, len(data))
	}

	// 失败之后进入冷却：一张取不到的图会被每个渲染它的 <img> 反复要一遍，
	// 一话几十页就是几十次无用请求。代价是上游在这几分钟内修好了也仍然报失败，
	// 这与海报代理的取舍一致，是有意的。
	recorder = httptest.NewRecorder()
	service.ServeHTTP(recorder, imageRequest("http://127.0.0.1:9/bad.png", "", ""))
	if recorder.Code != http.StatusServiceUnavailable {
		t.Fatalf("冷却期内的第二次请求 status = %d, want 503", recorder.Code)
	}
	if len(*requests) != 1 {
		t.Fatalf("冷却期内又出网了：upstream 收到 %d 次请求，want 1", len(*requests))
	}
}

func TestImageProxyServesSecondCallFromDisk(t *testing.T) {
	allowPrivateForTest(t)
	dir := t.TempDir()
	service := NewImageProxy(func() string { return dir })
	requests := stubImageUpstream(t, service, http.StatusOK, "image/png", tinyPNG)

	const upstream = "http://127.0.0.1:9/page-1.png"
	for i := 0; i < 2; i++ {
		recorder := httptest.NewRecorder()
		service.ServeHTTP(recorder, imageRequest(upstream, "", ""))
		if recorder.Code != http.StatusOK {
			t.Fatalf("call %d: status = %d", i, recorder.Code)
		}
		if !bytes.Equal(recorder.Body.Bytes(), tinyPNG) {
			t.Fatalf("call %d: 字节与上游不一致", i)
		}
	}
	// 计数是这条测试唯一要紧的断言：磁盘缓存有没有用，只看第二次还在不在出网。
	if len(*requests) != 1 {
		t.Fatalf("upstream got %d requests for 2 calls, want 1 (第二次应当命中磁盘缓存)", len(*requests))
	}
}

func TestImageProxyUsesItsOwnCacheDirectory(t *testing.T) {
	allowPrivateForTest(t)
	dir := t.TempDir()
	reading := NewImageProxy(func() string { return dir })
	posters := NewService(func() string { return dir })

	stubImageUpstream(t, reading, http.StatusOK, "image/png", tinyPNG)
	recorder := httptest.NewRecorder()
	reading.ServeHTTP(recorder, imageRequest("http://127.0.0.1:9/shared.png", "", ""))
	if recorder.Code != http.StatusOK {
		t.Fatalf("status = %d", recorder.Code)
	}

	// 两份缓存必须在不同子目录：淘汰是按 mtime 全局排序的，
	// 混在一个目录里，一话几十张大图会把开屏要用的小海报全挤出去。
	if reading.store.root() == posters.images.root() {
		t.Fatalf("阅读图片与海报共用了同一个缓存目录 %s", reading.store.root())
	}
	if !strings.HasSuffix(reading.store.root(), "readingcache") {
		t.Fatalf("阅读缓存目录 = %s, want …/readingcache", reading.store.root())
	}
}

func TestDecodeRefererRejectsNonHTTP(t *testing.T) {
	// Referer 是一个纯字符串头，不会被用来发请求，所以它不需要过 ValidateTarget；
	// 但它仍然必须是个 http(s) 地址——否则源站收到的会是一个 javascript: 之类的
	// 无意义值，而排查时会以为是防盗链的问题。
	for _, raw := range []string{"javascript:alert(1)", "file:///etc/passwd", "data:text/html,x", "not a url", ""} {
		encoded := base64.RawURLEncoding.EncodeToString([]byte(raw))
		if got := decodeReferer(encoded); got != "" {
			t.Fatalf("decodeReferer(%q) = %q, want empty", raw, got)
		}
	}
	if got := decodeReferer(base64.RawURLEncoding.EncodeToString([]byte("https://a.example.test/p?q=1#f"))); got != "https://a.example.test/p?q=1#f" {
		t.Fatalf("decodeReferer dropped part of a valid address: %q", got)
	}
	// 超长直接忽略：一个几 KB 的 Referer 没有意义，而它会占满请求行。
	if got := decodeReferer(base64.RawURLEncoding.EncodeToString([]byte(strings.Repeat("a", maxReadingRefBytes+10)))); got != "" {
		t.Fatalf("an over-long referer was accepted (%d bytes)", len(got))
	}
}
