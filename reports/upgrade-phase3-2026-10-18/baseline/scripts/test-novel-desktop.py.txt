"""Actual App/reading strategy in desktop Chrome with isolated native bindings."""
import argparse
import json
import os
import re
import subprocess
import time
from pathlib import Path
from urllib.request import urlopen
from playwright.sync_api import sync_playwright

parser=argparse.ArgumentParser();parser.add_argument('--mode',choices=['novel','comic'],default='novel');args=parser.parse_args()
mode=args.mode;port='19659' if mode=='comic' else '19658'
ROOT=Path(__file__).resolve().parent.parent
OUT=ROOT/('dist/audit/'+mode+'-desktop');OUT.mkdir(parents=True,exist_ok=True)
origin='http://127.0.0.1:'+port
server=subprocess.Popen(['node','node_modules/vite/bin/vite.js','--host','127.0.0.1','--port',port,'--strictPort'],cwd=ROOT/'frontend',stdout=subprocess.DEVNULL,stderr=subprocess.DEVNULL,creationflags=subprocess.CREATE_NO_WINDOW if os.name=='nt' else 0)
try:
 for _ in range(100):
  if server.poll() is not None:raise RuntimeError('Test Vite exited')
  try:
   with urlopen(origin+'/tests/browser/novel.html',timeout=1):break
  except OSError:time.sleep(.1)
 names=re.findall(r'export function (\w+)\(', (ROOT/'frontend/bindings/cczjVideo/app/service/app.js').read_text(encoding='utf-8'))
 version=re.search(r"LICENSE_TERMS_VERSION\s*=\s*'([^']+)'",(ROOT/'frontend/src/stores/licenseState.ts').read_text(encoding='utf-8')).group(1)
 mock='\n'.join('export async function '+name+'(...args){return '+('[]' if name.startswith(('List','GetAll','GetSources','GetDownload')) else 'false' if name.startswith('WindowIs') else 'null')+'}' for name in names if name not in ['GetSetting','SetSetting','GetAppVersion'])
 mock+='\nexport async function GetSetting(k){if(k===\'application_mode\')return sessionStorage.getItem(\'test-mode\')||\''+mode+'\';if(k===\'license_terms_version\')return '+json.dumps(version)+";return ''}"
 mock+="\nexport async function SetSetting(k,v){if(k==='application_mode')sessionStorage.setItem('test-mode',v)}\nexport async function GetAppVersion(){return '2.3.2'}"
 runtime="export const isMobileRuntime=()=>false,isDesktopRuntime=()=>true;export const Window={};export const Dialogs={};export const initializeRuntimeEnvironment=async()=>{};export function reloadApplication(url){location.replace(url)}"
 with sync_playwright() as pw:
  browser=pw.chromium.launch(headless=True,channel='chrome')
  try:
   page=browser.new_page(viewport={'width':1280,'height':900});errors=[]
   page.on('pageerror',lambda error:errors.append(str(error)))
   page.add_init_script("window.chrome ||= {};window.chrome.webview={postMessage(){},addEventListener(){}}")
   page.route('**/bindings/cczjVideo/app/service/app.js*',lambda route:route.fulfill(status=200,content_type='text/javascript',body=mock))
   page.route('**/src/api/runtime.ts*',lambda route:route.fulfill(status=200,content_type='text/javascript',body=runtime))
   page.goto(origin+'/tests/browser/novel.html#/'+mode)
   page.wait_for_selector('.'+mode+'-featured');page.wait_for_selector('.splash-wrap',state='detached')
   for width in [960,1280,1440]:
    page.set_viewport_size({'width':width,'height':900})
    views=[('home','/novel','.novel-featured'),('book','/novel/book/long-wind','.novel-detail-actions'),('reader','/novel/read/long-wind/0','.novel-reading-paragraph'),('settings','/novel/settings','.novel-settings')] if mode=='novel' else [('home','/comic','.comic-featured'),('series','/comic/series/moon-mail','.comic-detail'),('reader','/comic/read/moon-mail/0','.comic-reader'),('settings','/comic/settings','.comic-settings')]
    for name,path,selector in views:
     page.evaluate('path=>location.hash=path',path);page.wait_for_selector(selector);page.wait_for_timeout(450)
     assert page.locator('.cczj-'+mode+'-shell').evaluate('e=>e.scrollWidth<=e.clientWidth+1')
     if name=='home':assert page.locator('.'+mode+'-desktop-nav').is_visible() and not page.locator('.'+mode+'-bottom-nav').is_visible()
     page.screenshot(path=str(OUT/f'{width}-{name}.png'))
   assert not errors,errors
   print('PASS: 12 desktop '+mode+' App views, navigation, reader/settings layouts and no runtime errors')
  finally:browser.close()
finally:
 server.terminate()
 try:server.wait(timeout=5)
 except subprocess.TimeoutExpired:server.kill()
