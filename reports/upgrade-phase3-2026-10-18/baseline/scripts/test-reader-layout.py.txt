"""Native paginated typography at narrow and tablet widths, including large text."""
import json,time
from pathlib import Path
from android_webview import AndroidWebView
OUT=Path(__file__).resolve().parent.parent/'dist/audit/reader-layout';OUT.mkdir(exist_ok=True)
PINIA="document.querySelector('#app').__vue_app__.config.globalProperties.$pinia"
SNAPSHOT="Object.fromEntries(Object.keys(localStorage).filter(k=>k.startsWith('cczj.')).map(k=>[k,localStorage.getItem(k)]))"
RESTORE="d=>{for(const[k,v]of Object.entries(d))v===null?localStorage.removeItem(k):localStorage.setItem(k,v);for(const k of Object.keys(localStorage))if(k.startsWith('cczj.')&&!(k in d))localStorage.removeItem(k)}"
with AndroidWebView() as p:
 original_mode=p.evaluate(PINIA+"._s.get('applicationMode').mode");original_route=p.evaluate('location.hash')
 saved=p.evaluate(SNAPSHOT)
 def wait(expr):
  for _ in range(250):
   try:
    if p.evaluate(expr):return
   except RuntimeError:pass
   time.sleep(.1)
  raise AssertionError(expr)
 def switch(mode):
  current=p.evaluate(PINIA+"._s.get('applicationMode').mode")
  if current==mode:return
  p.evaluate('location.hash='+json.dumps('/'+current+'/settings' if current!='video' else '/settings?group=basic&section=appearance'))
  wait("!!document.querySelector('.mode-switch-options')")
  p.evaluate("document.querySelector('.mode-switch-options button:nth-child("+str(['video','novel','comic'].index(mode)+1)+")').click()")
  wait("!!document.querySelector('.app-shell[data-application-mode="+mode+"]') && !document.querySelector('.splash-wrap')")
 try:
  switch('novel')
  p.evaluate("location.hash='/novel/read/long-wind/0'");wait("!!document.querySelector('.novel-reader')")
  for width in [320,360,412,768]:
   p.call('Emulation.setDeviceMetricsOverride',{'width':width,'height':700,'deviceScaleFactor':1,'mobile':True})
   for size in [16,28]:
    p.evaluate(PINIA+"._s.get('novelReading').setPreferences({size:"+str(size)+",flow:'paged',margin:24,leading:1.85})")
    time.sleep(.6)
    state=p.evaluate("(()=>{const v=document.querySelector('.novel-page-viewport'),a=document.querySelector('.novel-pagination-track'),r=v.getBoundingClientRect();let clipped=0;const walker=document.createTreeWalker(a,NodeFilter.SHOW_TEXT);let n;while(n=walker.nextNode()){if(!n.textContent.trim())continue;const range=document.createRange();range.selectNodeContents(n);for(const b of range.getClientRects()){if(b.width&&b.right>r.left+.5&&b.left<r.right-.5&&(b.top<r.top-1||b.bottom>r.bottom+1))clipped++}}return{width:v.clientWidth,body:a.scrollWidth,clipped,overflow:document.querySelector('.novel-main').scrollWidth>innerWidth+1,status:document.querySelector('.novel-reading-status').textContent}})()")
    assert not state['overflow'] and not state['clipped'],(width,size,state)
    p.screenshot(OUT/f'novel-{width}-{size}.png')
  # The dialog supports native back, focus containment and keyboard dismissal.
  p.evaluate("document.querySelector('.novel-page-viewport').click()")
  p.evaluate("document.querySelector('.novel-reader-bottom nav button:last-child').click()")
  wait("!!document.querySelector('.reader-sheet')")
  p.call('Input.dispatchKeyEvent',{'type':'keyDown','key':'Tab','code':'Tab','windowsVirtualKeyCode':9,'modifiers':8})
  assert p.evaluate("!!document.activeElement.closest('.reader-sheet')")
  p.call('Input.dispatchKeyEvent',{'type':'keyDown','key':'Escape','code':'Escape','windowsVirtualKeyCode':27})
  wait("!document.querySelector('.reader-sheet')")
  print('PASS: actual text fragments fit 8 narrow/tablet typography layouts; dialog focus/back/keyboard behavior')
 finally:
  p.call('Emulation.clearDeviceMetricsOverride')
  p.evaluate("location.hash='/novel/settings'");wait("!!document.querySelector('.novel-settings')")
  p.evaluate('('+RESTORE+')('+json.dumps(saved)+')')
  switch(original_mode);p.evaluate('location.hash='+json.dumps(original_route));p.call('Page.reload')
