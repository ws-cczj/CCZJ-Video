 def wait(expr,timeout=30):
  end=time.monotonic()+timeout
  while time.monotonic()<end:
   try:
    if p.evaluate(expr):return
   except RuntimeError:pass
   time.sleep(.1)
  raise AssertionError(expr)
 def visit(path,selector):

 def click(selector):p.evaluate('document.querySelector('+json.dumps(selector)+').click()');time.sleep(.25)
 def text_click(selector,text):p.evaluate('Array.from(document.querySelectorAll('+json.dumps(selector)+')).find(e=>e.textContent.trim()==='+json.dumps(text)+').click()');time.sleep(.25)
 def touch(kind,x=0,y=0):p.call('Input.dispatchTouchEvent',{'type':kind,'touchPoints':[] if kind in ('touchEnd','touchCancel') else [{'x':x,'y':y}]})
 def point(selector,x=.5,y=.5):return p.evaluate('(()=>{const r=document.querySelector('+json.dumps(selector)+').getBoundingClientRect();return{x:r.x+r.width*'+str(x)+',y:r.y+r.height*'+str(y)+'}})()')
 def tap(selector,x=.5,y=.5):
  at=point(selector,x,y);touch('touchStart',**at);touch('touchEnd');time.sleep(.3)
 def swipe(selector,reverse=False,cancel=False):
  a=point(selector,.25 if reverse else .8,.45);b=point(selector,.8 if reverse else .25,.45)
  touch('touchStart',**a)
  for i in range(1,9):touch('touchMove',a['x']+(b['x']-a['x'])*i/8,a['y']);time.sleep(.025)
  touch('touchCancel' if cancel else 'touchEnd');time.sleep(.4)
 def back():subprocess.run(p.adb+['shell','input','keyevent','KEYCODE_BACK'],capture_output=True,check=True);time.sleep(.4)
